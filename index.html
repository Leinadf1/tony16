const M3U_FILE = "./lista.m3u"; async function init() { const video = document.getElementById('video'); const ui = video['ui']; const player = ui.getControls().getPlayer(); const mainList = document.getElementById('main-list'); const chanCount = document.getElementById('chan-count'); player.configure({ streaming: { jumpLargeGaps: true } }); try { const response = await fetch(M3U_FILE + "?t=" + Date.now()); const text = await response.text(); // Debug in console per vedere cosa sta leggendo davvero console.log("Contenuto grezzo ricevuto:", text.substring(0, 100)); // Pulizia del testo da caratteri invisibili (BOM) const cleanText = text.replace(/^\uFEFF/, ''); const lines = cleanText.split(/\r?\n/); let current = { drm: {}, group: "ALTRI", name: "", logo: "" }; let count = 0; mainList.innerHTML = ""; for (let line of lines) { line = line.trim(); if (!line) continue; // Salta righe vuote // Cerca Chiavi DRM (formato kodi o semplice) if (line.toLowerCase().includes('license_key=')) { let val = line.split('=')[1] || ""; val = val.replace(/["']/g, ""); // Rimuove eventuali virgolette if (val.startsWith('{')) { try { current.drm = JSON.parse(val); } catch(e) {} } else if (val.includes(':')) { let parts = val.split(':'); current.drm[parts[0].trim()] = parts[1].trim(); } } // Cerca Info Canale else if (line.startsWith('#EXTINF:')) { let logoMatch = line.match(/tvg-logo="([^"]+)"/i); let groupMatch = line.match(/group-title="([^"]+)"/i); let nameMatch = line.match(/,(.*)$/); if (logoMatch) current.logo = logoMatch[1]; if (groupMatch) current.group = groupMatch[1]; if (nameMatch) current.name = nameMatch[1].trim(); } // Se la riga contiene un link, chiudi il canale e aggiungilo else if (line.toLowerCase().startsWith('http')) { current.url = line; // Se non ha nome, usa una parte dell'URL if (!current.name) current.name = "Canale " + (count + 1); let gridId = `grid-${current.group.toLowerCase().replace(/[^a-z0-9]/g, '-')}`; let grid = document.getElementById(gridId); if (!grid) { const container = document.createElement('div'); container.className = 'category-container'; container.innerHTML = `
${current.group.toUpperCase()}
`; mainList.appendChild(container); grid = document.getElementById(gridId); } const card = document.createElement('div'); card.className = 'channel-card'; const safeDrm = JSON.stringify(current.drm); const safeUrl = current.url; card.innerHTML = `
${current.name}
`; // Usiamo una closure per bloccare i dati correnti (function(url, drm){ card.onclick = async () => { await player.unload(); player.configure({ drm: { clearKeys: drm } }); await player.load(url); window.scrollTo({top: 0, behavior: 'smooth'}); }; })(safeUrl, current.drm); grid.appendChild(card); count++; current = { drm: {}, group: "ALTRI", name: "", logo: "" }; } } if (count === 0) { chanCount.innerText = "ERRORE: FORMATO M3U NON RICONOSCIUTO"; } else { chanCount.innerText = count + " CANALI CARICATI"; } } catch (e) { chanCount.innerText = "ERRORE DI RETE O FILE MANCANTE"; } } document.addEventListener('shaka-ui-loaded', init);
